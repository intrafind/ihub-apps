## 2. Fixture helper — `tests/utils/officeFixtures.js` (new)

Validated in the spike (scratch copy); adapt to the repo's lint rules.

```js
import JSZip from 'jszip';

const W_NS =
  'xmlns:w="http://schemas.openxmlformats.org/wordprocessingml/2006/main" ' +
  'xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships"';

/** jsdom's File may lack arrayBuffer(); the extractors only need name/type/arrayBuffer. */
export function makeFile(arrayBuffer, name, type) {
  const file = new File([arrayBuffer], name, { type });
  if (typeof file.arrayBuffer !== 'function') file.arrayBuffer = async () => arrayBuffer;
  return file;
}

/** One paragraph; `ppr` / `rpr` are raw inner XML of w:pPr / w:rPr. */
export const p = (text, ppr = '', rpr = '') =>
  `<w:p>${ppr ? `<w:pPr>${ppr}</w:pPr>` : ''}<w:r>${rpr ? `<w:rPr>${rpr}</w:rPr>` : ''}` +
  `<w:t xml:space="preserve">${text}</w:t></w:r></w:p>`;

/** A numbering level definition for numbering.xml. */
export const lvl = (ilvl, numFmt, lvlText, start = 1) =>
  `<w:lvl w:ilvl="${ilvl}"><w:start w:val="${start}"/><w:numFmt w:val="${numFmt}"/>` +
  `<w:lvlText w:val="${lvlText}"/></w:lvl>`;

export const numPr = (numId, ilvl = 0) =>
  `<w:numPr><w:ilvl w:val="${ilvl}"/><w:numId w:val="${numId}"/></w:numPr>`;

/**
 * Minimal DOCX from raw WordprocessingML: `body` (w:body content), `styles`
 * (w:style elements), `numbering` (w:abstractNum / w:num), optional extra parts
 * such as { 'word/header1.xml': '…' } plus `rels` entries for them.
 * Returns a File whose arrayBuffer was produced in the test realm.
 */
export async function buildDocxFile({ body, styles = '', numbering = '', parts = {}, rels = '' },
  name = 'test.docx') {
  const zip = new JSZip();
  zip.file('[Content_Types].xml',
    '<?xml version="1.0" encoding="UTF-8" standalone="yes"?><Types xmlns="http://schemas.openxmlformats.org/package/2006/content-types">' +
    '<Default Extension="rels" ContentType="application/vnd.openxmlformats-package.relationships+xml"/>' +
    '<Default Extension="xml" ContentType="application/xml"/>' +
    '<Override PartName="/word/document.xml" ContentType="application/vnd.openxmlformats-officedocument.wordprocessingml.document.main+xml"/>' +
    '<Override PartName="/word/styles.xml" ContentType="application/vnd.openxmlformats-officedocument.wordprocessingml.styles+xml"/>' +
    '<Override PartName="/word/numbering.xml" ContentType="application/vnd.openxmlformats-officedocument.wordprocessingml.numbering+xml"/>' +
    '</Types>');
  zip.file('_rels/.rels',
    '<?xml version="1.0" encoding="UTF-8" standalone="yes"?><Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">' +
    '<Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/officeDocument" Target="word/document.xml"/></Relationships>');
  zip.file('word/_rels/document.xml.rels',
    '<?xml version="1.0" encoding="UTF-8" standalone="yes"?><Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">' +
    '<Relationship Id="rIdS" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/styles" Target="styles.xml"/>' +
    '<Relationship Id="rIdN" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/numbering" Target="numbering.xml"/>' +
    `${rels}</Relationships>`);
  zip.file('word/styles.xml', `<?xml version="1.0" encoding="UTF-8" standalone="yes"?><w:styles ${W_NS}>${styles}</w:styles>`);
  zip.file('word/numbering.xml', `<?xml version="1.0" encoding="UTF-8" standalone="yes"?><w:numbering ${W_NS}>${numbering}</w:numbering>`);
  zip.file('word/document.xml', `<?xml version="1.0" encoding="UTF-8" standalone="yes"?><w:document ${W_NS}><w:body>${body}</w:body></w:document>`);
  for (const [path, xml] of Object.entries(parts)) zip.file(path, xml);
  const arrayBuffer = await zip.generateAsync({ type: 'arraybuffer' });
  return makeFile(arrayBuffer, name, 'application/vnd.openxmlformats-officedocument.wordprocessingml.document');
}
```

## T-PERF-01 method (measured, 2026-10-08)

Timing in jest/jsdom is useless for this: jsdom's DOM is an order of magnitude slower than a
browser's and hides the one defect that matters. The numbers below come from real Chromium:

1. Put a temporary entry (generate a DOCX with `tests/utils/officeFixtures.js`, run legacy and
   structured extraction five times, report medians) in `client/src/`, bundle it with Vite's
   library build (`build.lib`, format `iife`, alias `jszip` to its `lib/index.js`) so mammoth,
   Turndown and JSZip resolve exactly as in the shipped client.
2. Load the bundle in the Playwright Chromium (`/opt/pw-browsers/chromium`, `--no-sandbox`) via
   `page.addScriptTag` and call the entry. Remove the entry afterwards; it is not committed.

Result for 5,000 paragraphs (headings, outline-level styles, 50 tables; 723 KB file):

| Version | Legacy | Structured | Ratio |
|---|---|---|---|
| First implementation (one Turndown call) | 148 ms | 1,102 ms | 7.4× |
| Blocks of 100 top-level elements per Turndown call | 140–150 ms | 290–320 ms | 1.9–2.2× (median 2.2×) |

With 20,000 paragraphs (2.9 MB) the first version took 12.6 s (21×, quadratic: Turndown copies
the whole output so far for every block); the chunked version takes 1.4 s (2.3×, linear). Phases
at 5,000 paragraphs: parse 19 ms, OOXML pass 40 ms, serialize 23 ms, zip 59 ms, mammoth 170 ms
(legacy: 150 ms), Markdown 55 ms. The floor is set by parsing the document a second time; trying
DEFLATE or `uint8array` output for the intermediate zip changed nothing measurable.
